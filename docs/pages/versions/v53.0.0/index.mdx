import React, { useState } from 'react';
import { View, Text, TextInput, Button, FlatList, Image, TouchableOpacity, StyleSheet, Picker } from 'react-native';

const ilanlar = [
  {
    id: '1',
    baslik: 'Modern Daire Milano’da',
    fiyat: 850,
    aciklama: '2+1, merkezi konumda, metroya yakın.',
    foto: 'https://picsum.photos/200/150?random=1',
    konum: 'Milano',
  },
  {
    id: '2',
    baslik: 'Şirin Oda Roma’da',
    fiyat: 500,
    aciklama: 'Tek kişilik oda, üniversiteye yakın.',
    foto: 'https://picsum.photos/200/150?random=2',
    konum: 'Roma',
  },
  // Daha fazla ilan eklenebilir
];

export default function AnaSayfa() {
  const [sehir, setSehir] = useState('');
  const [filtreliIlanlar, setFiltreliIlanlar] = useState(ilanlar);

  const sehirFiltrele = (secilenSehir) => {
    setSehir(secilenSehir);
    if (secilenSehir === '') {
      setFiltreliIlanlar(ilanlar);
    } else {
      setFiltreliIlanlar(ilanlar.filter(i => i.konum === secilenSehir));
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.baslik}>CasaVeloce</Text>

      <Picker
        selectedValue={sehir}
        onValueChange={(itemValue) => sehirFiltrele(itemValue)}
        style={styles.picker}
      >
        <Picker.Item label="Şehir seç" value="" />
        <Picker.Item label="Milano" value="Milano" />
        <Picker.Item label="Roma" value="Roma" />
      </Picker>

      <FlatList
        data={filtreliIlanlar}
        keyExtractor={item => item.id}
        renderItem={({ item }) => (
          <View style={styles.ilanKart}>
            <Image source={{ uri: item.foto }} style={styles.foto} />
            <View style={styles.ilanDetay}>
              <Text style={styles.ilanBaslik}>{item.baslik}</Text>
              <Text>{item.aciklama}</Text>
              <Text style={styles.fiyat}>{item.fiyat} € / ay</Text>
              <TouchableOpacity style={styles.detayButon}>
                <Text style={{color: '#fff'}}>Detaylar</Text>
              </TouchableOpacity>
            </View>
          </View>
        )}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex:1, padding: 20, backgroundColor:'#fff' },
  baslik: { fontSize: 28, fontWeight: 'bold', marginBottom: 15 },
  picker: { height: 50, marginBottom: 20 },
  ilanKart: { flexDirection: 'row', marginBottom: 20, borderColor:'#ddd', borderWidth:1, borderRadius: 8, overflow: 'hidden' },
  foto: { width: 120, height: 90 },
  ilanDetay: { flex: 1, padding: 10 },
  ilanBaslik: { fontSize: 18, fontWeight: 'bold' },
  fiyat: { marginTop: 8, fontWeight: 'bold', color: '#2a9d8f' },
  detayButon: { marginTop: 10, backgroundColor: '#264653', padding: 8, borderRadius: 5, alignItems: 'center' }
})
